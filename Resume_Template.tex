% HOW TO USE
% - Fill in the bracketed [placeholder] text throughout.
% - \resumeSubheading{Company}{Location}{Title}{Dates} starts a job entry.
%   Follow it with \resumeItemListStart ... \resumeItemListEnd containing
%   one \resumeItem{...} per bullet.
% - If you were promoted within the same company, use \resumeSubheading
%   for the most recent title, then a plain \resumeSubSubheading{Title}{Dates}
%   for the earlier title underneath it (no company/location repeated),
%   each with its own \resumeItemListStart/End block of bullets.
% - \resumeProjectHeading works the same way for the Projects section.
% - Compile with: pdflatex resume_template.tex
%-------------------------

% Custom commands -- do not edit below this line
\newcommand{\resumeItem}[1]{
  \item\small{
    {#1 \vspace{-2pt}}
  }
}

\newcommand{\resumeSubheading}[4]{
  \vspace{-2pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeSubSubheading}[2]{
    \item
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
      \textit{\small#1} & \textit{\small #2} \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeProjectHeading}[2]{
    \item
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
      \small#1 & #2 \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeSubItem}[1]{\resumeItem{#1}\vspace{-4pt}}
\renewcommand\labelitemii{$\vcenter{\hbox{\tiny$\bullet$}}$}
\newcommand{\resumeSubHeadingListStart}{\begin{itemize}[leftmargin=0.15in, label={}]}
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}}
\newcommand{\resumeItemListStart}{\begin{itemize}}
\newcommand{\resumeItemListEnd}{\end{itemize}\vspace{-5pt}}

%---------- EXPERIENCE ----------
\section{Experience}
  \resumeSubHeadingListStart

    % --- Example: a role where you were promoted within the same company ---
    \resumeSubheading
      {[Company Name]}{[City, State / Remote]}
      {[Most Recent Title]}{[Start Date] -- [End Date / Present]}
      \resumeItemListStart
        \resumeItem{[Bullet: lead with a strong action verb, describe what you did, and quantify the result wherever possible -- e.g. "Owned X end-to-end, resulting in Y\% improvement in Z."]}
        \resumeItem{[Bullet 2 -- another concrete accomplishment with a measurable outcome.]}
        \resumeItem{[Bullet 3 -- vary the type of accomplishment: technical build, process improvement, cross-functional collaboration, etc.]}
      \resumeItemListEnd

    \resumeSubSubheading
      {[Earlier Title at Same Company]}{[Start Date] -- [End Date]}
      \resumeItemListStart
        \resumeItem{[Bullet describing a core responsibility or accomplishment in this earlier role.]}
        \resumeItem{[Bullet 2.]}
      \resumeItemListEnd

    % --- Example: a standard single-title role ---
    \resumeSubheading
      {[Previous Company Name]}{[City, State]}
      {[Job Title]}{[Start Date] -- [End Date]}
      \resumeItemListStart
        \resumeItem{[Bullet describing scope and impact in this role.]}
      \resumeItemListEnd

  \resumeSubHeadingListEnd
